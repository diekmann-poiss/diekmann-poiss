        <div class="whyus-body">

            <p class="whyus-lead reveal">We combine hands-on implementation experience with scientific rigour — two disciplines that rarely sit together in the same room.</p>

            <p class="whyus-text reveal reveal-delay-1">Diekmann & Poiss grew out of years spent at the intersection of landscape architecture, building physics, and data science. We have planned, built, and monitored living wall and green roof systems across a range of building typologies — and used those projects as laboratories for the tools we now offer to clients.</p>

            <p class="whyus-text reveal reveal-delay-2">Our approach is rooted in measurable outcomes: thermal comfort, plant survival rates, water efficiency, long-term maintenance cost. Every recommendation we make is one we can back with data — because in most cases we have collected that data ourselves.</p>

            <p class="whyus-text reveal reveal-delay-3">We are a small practice by design. Every project is handled directly by us, not delegated. That means consistent quality, fast turnaround, and a single point of contact from first consultation to final handover.</p>

    <script>

        function showPage(pageId) {

            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));

            document.getElementById(pageId).classList.add('active');

            window.scrollTo(0, 0);

            observeReveals();

        }


        window.addEventListener('scroll', () => {

            const bg = document.querySelector('.hero-bg');

            if (!bg) return;

            const heroH = document.getElementById('hero').offsetHeight;

            const y = window.scrollY;

            if (y <= heroH) bg.style.transform = `translateY(${y * 0.3}px)`;

        }, { passive: true });


        function observeReveals() {

            const observer = new IntersectionObserver((entries) => {

                entries.forEach(e => {

                    if (e.isIntersecting) { e.target.classList.add('visible'); observer.unobserve(e.target); }

                });

            }, { threshold: 0.12 });

            document.querySelectorAll('.reveal:not(.visible)').forEach(el => observer.observe(el));

        }


        function scrollToSection(sectionId) {

            const section = document.getElementById(sectionId);

            if (section) {

                window.scrollTo({ top: section.offsetTop - 80, behavior: 'smooth' });

                setTimeout(observeReveals, 500);

            }

        }


        window.addEventListener('scroll', () => {

            const scrollPosition = window.scrollY + 200;

            const sections = ['service-structure','service-soil','service-plants','service-irrigation','service-maintenance'];

            let activeIndex = -1;

            sections.forEach((id, index) => {

                const section = document.getElementById(id);

                if (section) {

                    const top = section.offsetTop;

                    if (scrollPosition >= top && scrollPosition < top + section.offsetHeight) activeIndex = index;

                }

            });

            sections.forEach((id, index) => {

                const node = document.getElementById(`node-${index + 1}`);

                if (node) node.classList.toggle('active', index === activeIndex);

            });

        });


        observeReveals();

    </script>
